cmake_minimum_required(VERSION 3.22.1)
project(sea_g2p_jni LANGUAGES CXX)
add_library(sea_g2p_rs SHARED IMPORTED)
set_target_properties(sea_g2p_rs PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/arm64-v8a/libsea_g2p_rs.so" IMPORTED_NO_SONAME TRUE)
add_library(sea_g2p_jni SHARED sea_g2p_jni.cpp)
target_include_directories(sea_g2p_jni PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_features(sea_g2p_jni PRIVATE cxx_std_17)
find_library(log-lib log)
target_link_libraries(sea_g2p_jni PRIVATE sea_g2p_rs ${log-lib})
